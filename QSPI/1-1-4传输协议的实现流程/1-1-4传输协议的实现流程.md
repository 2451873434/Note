# 1-1-4传输协议的实现流程

> ​	`1-1-4`传输协议分为`1-1-4`读传输协议和`1-1-4`写传输协议。
>
> ​	目前只有`6.6`版本内核同时支持`1-1-4`读传输协议和`1-1-4`写传输协议，`5.15`及以下版本内核只支持`1-1-4`读传输协议。
>
> ​	值得一提的是，虽然`6.6`版本内核的`flash`驱动支持`1-1-4`写传输协议，但是大多数型号的`flash`描述中只设置了`1-1-4`读传输协议，而没有设置`1-1-4`写传输协议。这一点会在本文后续进行解释。



## 1、不同型号`flash`描述

​	以`gd25q128`为例。

```c
/* drivers/mtd/spi-nor/gigadevice.c */
static const struct flash_info gigadevice_nor_parts[] = {
    { "gd25q128", INFO(0xc84018, 0, 64 * 1024, 256)
	FLAGS(SPI_NOR_HAS_LOCK | SPI_NOR_HAS_TB)
	NO_SFDP_FLAGS(SECT_4K | SPI_NOR_DUAL_READ |
		      SPI_NOR_QUAD_READ) },
};
```

​	通过`SPI_NOR_QUAD_READ`可知该型号的`flash`设置了`1-1-4`读传输协议，我们索引`SPI_NOR_QUAD_READ`宏可以找到与之对应的宏`SPI_NOR_QUAD_PP`，如下。

```c
/* driver/mtd/spi-nor/core.h */
u16 flags;
#define SPI_NOR_HAS_LOCK		BIT(0)
#define SPI_NOR_HAS_TB			BIT(1)
#define SPI_NOR_TB_SR_BIT6		BIT(2)
#define SPI_NOR_4BIT_BP			BIT(3)
#define SPI_NOR_BP3_SR_BIT6		BIT(4)
#define SPI_NOR_SWP_IS_VOLATILE		BIT(5)
#define SPI_NOR_NO_ERASE		BIT(6)
#define NO_CHIP_ERASE			BIT(7)
#define SPI_NOR_NO_FR			BIT(8)
#define SPI_NOR_QUAD_PP			BIT(9)
#define SPI_NOR_RWW			BIT(10)

	u8 no_sfdp_flags;
#define SPI_NOR_SKIP_SFDP		BIT(0)
#define SECT_4K				BIT(1)
#define SPI_NOR_DUAL_READ		BIT(3)
#define SPI_NOR_QUAD_READ		BIT(4)
#define SPI_NOR_OCTAL_READ		BIT(5)
#define SPI_NOR_OCTAL_DTR_READ		BIT(6)
#define SPI_NOR_OCTAL_DTR_PP		BIT(7)
```

> 注意，`SPI_NOR_QUAD_READ`为`no_sfdp_flags`的值，`SPI_NOR_QUAD_PP`为`flags`的值，与之对应的是`NO_SFDP_FLAGS`和`FLAGS`。
>
> `5.15`及以下版本的内核没有`SPI_NOR_QUAD_PP`的定义，所以`5.15`及以下版本内核只支持`1-1-4`读传输协议。

## 2、`flash`驱动（core层）

### 2.1 是否支持`1-1-4`传输协议

​	在`flash`驱动注册时会做一些初始化参数的操作，而`flash`驱动是否支持`1-1-4`协议就是在初始化参数时设置，可以发现如下代码：

**`1-1-4`读传输协议**

```c
/* drivers/mtd/spi-nor/core.c */
void spi_nor_set_read_settings(struct spi_nor_read_command *read,
			  u8 num_mode_clocks,
			  u8 num_wait_states,
			  u8 opcode,
			  enum spi_nor_protocol proto)
{
	read->num_mode_clocks = num_mode_clocks;
	read->num_wait_states = num_wait_states;
	read->opcode = opcode;
	read->proto = proto;
}

static void spi_nor_init_default_params(struct spi_nor *nor)
{
    /* ... */
    params->hwcaps.mask |= SNOR_HWCAPS_READ;
	spi_nor_set_read_settings(&params->reads[SNOR_CMD_READ],
				  0, 0, SPINOR_OP_READ,
				  SNOR_PROTO_1_1_1);
    /* ... */
}

static void spi_nor_no_sfdp_init_params(struct spi_nor *nor)
{
    	if (no_sfdp_flags & SPI_NOR_QUAD_READ) {
		params->hwcaps.mask |= SNOR_HWCAPS_READ_1_1_4;
		spi_nor_set_read_settings(&params->reads[SNOR_CMD_READ_1_1_4],
					  0, 8, SPINOR_OP_READ_1_1_4,
					  SNOR_PROTO_1_1_4);
		}
}
```

​	由上述代码可以发现，`flash`驱动默认支持`1-1-1`的读传输协议，当通过第26行条件判断时，才支持`1-1-4`的读传输协议。而第26行的条件判断中的`no_sfdp_flags`参数就是第一节提到的`NO_SFDP_FLAGS`。上述的`gd25q128`的`NO_SFDP_FLAGS`为`NO_SFDP_FLAGS(SECT_4K | SPI_NOR_DUAL_READ | SPI_NOR_QUAD_READ)`，所以其支持`1-1-4`读传输协议。

​	此外，通过`spi_nor_set_read_settings()`函数会将读指令、`dummy`和传输协议存放进`&params->reads[x]`中，这样便于在后续代码中通过索引`x`获取到对应的读指令、`dummy`和传输协议（`dummy = num_mode_clocks + num_wait_states`）。读索引如下：

```c
enum spi_nor_read_command_index {
	SNOR_CMD_READ,
	SNOR_CMD_READ_FAST,
	SNOR_CMD_READ_1_1_1_DTR,

	/* Dual SPI */

	SNOR_CMD_READ_1_2_2,
	SNOR_CMD_READ_2_2_2,
	SNOR_CMD_READ_1_2_2_DTR,

	/* Quad SPI */
	SNOR_CMD_READ_1_1_4,
	SNOR_CMD_READ_1_4_4,
	SNOR_CMD_READ_4_4_4,
	SNOR_CMD_READ_1_4_4_DTR,

	/* Octal SPI */
	SNOR_CMD_READ_1_1_8,
	SNOR_CMD_READ_1_8_8,
	SNOR_CMD_READ_8_8_8,
	SNOR_CMD_READ_1_8_8_DTR,
	SNOR_CMD_READ_8_8_8_DTR,

	SNOR_CMD_READ_MAX
};
```



**`1-1-4`写传输协议**

```c
/* drivers/mtd/spi-nor/core.c */
void spi_nor_set_pp_settings(struct spi_nor_pp_command *pp, u8 opcode,
			     enum spi_nor_protocol proto)
{
	pp->opcode = opcode;
	pp->proto = proto;
}

static void spi_nor_init_default_params(struct spi_nor *nor)
{
    /* ... */
    /* Page Program settings. */
	params->hwcaps.mask |= SNOR_HWCAPS_PP;
	spi_nor_set_pp_settings(&params->page_programs[SNOR_CMD_PP],
				SPINOR_OP_PP, SNOR_PROTO_1_1_1);

	if (info->flags & SPI_NOR_QUAD_PP) {
		params->hwcaps.mask |= SNOR_HWCAPS_PP_1_1_4;
		spi_nor_set_pp_settings(&params->page_programs[SNOR_CMD_PP_1_1_4],
					SPINOR_OP_PP_1_1_4, SNOR_PROTO_1_1_4);
	}
}
```

​	由上述代码可以发现，`flash`驱动默认支持`1-1-1`的写传输协议，当通过第17行条件判断时，才支持`1-1-4`的写传输协议。而第17行的条件判断中的`info->flags`参数就是第一节提到的`FLAGS`。上述的`gd25q128`的`FLGAS`为`FLAGS(SPI_NOR_HAS_LOCK | SPI_NOR_HAS_TB)`，所以其不支持`1-1-4`写传输协议。

​	此外，通过`spi_nor_set_pp_settings()`函数会将写指令和传输协议存放进`&params->page_programs[x]`中，这样便于在后续代码中通过索引`x`获取到对应的写指令和传输协议。写索引如下：

```c
enum spi_nor_pp_command_index {
	SNOR_CMD_PP,

	/* Quad SPI */
	SNOR_CMD_PP_1_1_4,
	SNOR_CMD_PP_1_4_4,
	SNOR_CMD_PP_4_4_4,

	/* Octal SPI */
	SNOR_CMD_PP_1_1_8,
	SNOR_CMD_PP_1_8_8,
	SNOR_CMD_PP_8_8_8,
	SNOR_CMD_PP_8_8_8_DTR,

	SNOR_CMD_PP_MAX
};
```

> ​	需要注意的是，`1-1-1`传输协议读数据没有延时，`1-1-4`传输协议读数据有8个`clock`的`dummy`。而不论是`1-1-1`还是`1-1-4`传输协议的写数据都没有延时，所以不需要保存`dummy`数据。

​	

​	有一个专门用来标志`flash`是否支持各个不同传输协议的参数`params->hwcaps.mask`，需要注意的是不论是读传输协议还是写传输协议，都是共用这一个参数。其中，读传输协议使用的是`bit[16:0]`，写传输协议使用的是`bit[23:16]`。且传输协议越复杂，其所在的`bit`位越高。

```c
/* 读传输协议 */

#define SNOR_HWCAPS_READ_1_1_4		BIT(7)
```

### 2.2 是否采用`1-1-4`传输协议

​	由`2.1`节中的内容可知，`gd25q128`支持`1-1-1`的读/写传输协议和`1-1-4`的读传输协议，那么它究竟采用哪种传输协议呢？其实，具体使用哪种传输协议是由设备树描述决定的。

```c
static void spi_nor_spimem_adjust_hwcaps(struct spi_nor *nor, u32 *hwcaps)
{
	struct spi_nor_flash_parameter *params = nor->params;
	unsigned int cap;

	/* X-X-X modes are not supported yet, mask them all. */
	*hwcaps &= ~SNOR_HWCAPS_X_X_X;

	/*
	 * If the reset line is broken, we do not want to enter a stateful
	 * mode.
	 */
	if (nor->flags & SNOR_F_BROKEN_RESET)
		*hwcaps &= ~(SNOR_HWCAPS_X_X_X | SNOR_HWCAPS_X_X_X_DTR);

	for (cap = 0; cap < sizeof(*hwcaps) * BITS_PER_BYTE; cap++) {
		int rdidx, ppidx;

		if (!(*hwcaps & BIT(cap)))
			continue;

		rdidx = spi_nor_hwcaps_read2cmd(BIT(cap));
		if (rdidx >= 0 &&
		    spi_nor_spimem_check_readop(nor, &params->reads[rdidx]))
			*hwcaps &= ~BIT(cap);

		ppidx = spi_nor_hwcaps_pp2cmd(BIT(cap));
		if (ppidx < 0)
			continue;

		if (spi_nor_spimem_check_pp(nor,
					    &params->page_programs[ppidx]))
			*hwcaps &= ~BIT(cap);
	}
}
```

​	第6、7行，可以发现`x-x-x`模式的传输协议（`2-2-2`、`4-4-4`、`8-8-8`）是不支持的，所以将其对应的`hwcaps`位清零。

​	第16-34行，基本逻辑如下：从低位到高位遍历`hwcaps`中每一位，过滤掉不为`1`的`bit`位（第19-20行），那么剩下的数值为`1`的`bit`位就表示所支持的传输协议，再然后校验所支持的传输协议是否合法（第22-33行），如果不合法，则将其对应的`hwcaps`位清零（第25、33行）。最后的得到的`hwcaps`参数才是真正表示支持的合法的传输协议。

​	第22-23行的校验就是将支持的合法传输协议和设备树里设置的`spi-rx-bus-width`和`spi-tx-bus-width`做对比，如果读传输协议最大的数大于`spi-rx-bus-width`，则其对应的`hwcaps`的`bit`位清零，反之，不做处理。比如，如果`spi-rx-bus-width`为2，则会将`hwcaps`的`1-1-4`对应的`bit`位清零，最后得到的`hwcaps`参数表示支持`1-1-1`、`1-1-2`读传输协议。

​	我们需要清楚一点，到此处只是知道`flash`最终支持哪些传输协议，至于最后`flash`驱动采用哪种协议，是通过`spi_nor_select_read()`和`spi_nor_select_pp()`函数来决定的。

```c
static int spi_nor_select_read(struct spi_nor *nor,
			       u32 shared_hwcaps)
{
	int cmd, best_match = fls(shared_hwcaps & SNOR_HWCAPS_READ_MASK) - 1;
	const struct spi_nor_read_command *read;

	if (best_match < 0)
		return -EINVAL;

	cmd = spi_nor_hwcaps_read2cmd(BIT(best_match));
	if (cmd < 0)
		return -EINVAL;

	read = &nor->params->reads[cmd];
	nor->read_opcode = read->opcode;
	nor->read_proto = read->proto;
	nor->read_dummy = read->num_mode_clocks + read->num_wait_states;
	return 0;
}
```

​	第4行，找到`shared_hwcaps`参数中和读有关的位（`bit[16:0]`）的最高位，认为是最好的传输协议。此处很好理解，前面提到，传输协议越复杂，其所在的`bit`位越高。那么找到最高位对应的读传输协议，则该传输协议最复杂，其传输速度也就最高，也就是最好的传输协议了（这里的参数也很有意思，`best_match`表示最好的匹配到的`bit`位）。第10行通过匹配到的传输协议对应的最好的`bit`位找到对应的索引号。第14行通过索引号找到之前存放有指令、`dummy`和传输协议的`&nor->params->reads[cmd]`参数。第15-17行实现真正的赋值，即最后采用的是哪种传输协议。

## 3、具体校验流程分析

​	重点就是在`spi_nor_spimem_check_read()`和`spi_nor_spimem_check_pp()`函数的校验传输协议是否合法的逻辑处理，下面进行详细讲解。

> ​	如果只是想清楚表层逻辑，则无需看下面的内容。只需要知道如下逻辑即可：`w25q128`支持`1-1-1`、`1-1-2`、`1-1-4`的读传输协议，如果设备树中设置`spi-rx-bus-width`的值为2，则`1-1-4`传输协议会被过滤掉，最终采用的是`1-1-2`传输协议。写传输协议逻辑与其相似。

​	本文以`1-1-4`读传输协议的校验举例，写传输协议的校验逻辑与其相似。

1. 首先，`hwcaps`的每一位都有对应的一个索引号，通过`spi_nor_hwcaps_read2cmd()`函数找到`1-1-4`对应的索引号`SNOR_CMD_READ_1_1_4`。如下：

```c
int spi_nor_hwcaps_read2cmd(u32 hwcaps)
{
	static const int hwcaps_read2cmd[][2] = {
		{ SNOR_HWCAPS_READ,		SNOR_CMD_READ },
		{ SNOR_HWCAPS_READ_FAST,	SNOR_CMD_READ_FAST },
		{ SNOR_HWCAPS_READ_1_1_1_DTR,	SNOR_CMD_READ_1_1_1_DTR },
		{ SNOR_HWCAPS_READ_1_1_2,	SNOR_CMD_READ_1_1_2 },
		{ SNOR_HWCAPS_READ_1_2_2,	SNOR_CMD_READ_1_2_2 },
		{ SNOR_HWCAPS_READ_2_2_2,	SNOR_CMD_READ_2_2_2 },
		{ SNOR_HWCAPS_READ_1_2_2_DTR,	SNOR_CMD_READ_1_2_2_DTR },
		{ SNOR_HWCAPS_READ_1_1_4,	SNOR_CMD_READ_1_1_4 },
		{ SNOR_HWCAPS_READ_1_4_4,	SNOR_CMD_READ_1_4_4 },
		{ SNOR_HWCAPS_READ_4_4_4,	SNOR_CMD_READ_4_4_4 },
		{ SNOR_HWCAPS_READ_1_4_4_DTR,	SNOR_CMD_READ_1_4_4_DTR },
		{ SNOR_HWCAPS_READ_1_1_8,	SNOR_CMD_READ_1_1_8 },
		{ SNOR_HWCAPS_READ_1_8_8,	SNOR_CMD_READ_1_8_8 },
		{ SNOR_HWCAPS_READ_8_8_8,	SNOR_CMD_READ_8_8_8 },
		{ SNOR_HWCAPS_READ_1_8_8_DTR,	SNOR_CMD_READ_1_8_8_DTR },
		{ SNOR_HWCAPS_READ_8_8_8_DTR,	SNOR_CMD_READ_8_8_8_DTR },
	};

	return spi_nor_hwcaps2cmd(hwcaps, hwcaps_read2cmd,
				  ARRAY_SIZE(hwcaps_read2cmd));
}
```

2. 通过索引号就可以获得`&params->read[rdidx]`参数，该参数中存放有读指令、`dummy`和传输协议。通过`spi_nor_spimem_check_readop()`函数对该参数进行校验。

```c

#define SPI_NOR_READ_OP(opcode)						\
	SPI_MEM_OP(SPI_MEM_OP_CMD(opcode, 0),				\
		   SPI_MEM_OP_ADDR(3, 0, 0),				\
		   SPI_MEM_OP_DUMMY(1, 0),				\
		   SPI_MEM_OP_DATA_IN(2, NULL, 0))

void spi_nor_spimem_setup_op(const struct spi_nor *nor,
			     struct spi_mem_op *op,
			     const enum spi_nor_protocol proto)
{
	u8 ext;

	op->cmd.buswidth = spi_nor_get_protocol_inst_nbits(proto);

	if (op->addr.nbytes)
		op->addr.buswidth = spi_nor_get_protocol_addr_nbits(proto);

	if (op->dummy.nbytes)
		op->dummy.buswidth = spi_nor_get_protocol_addr_nbits(proto);

	if (op->data.nbytes)
		op->data.buswidth = spi_nor_get_protocol_data_nbits(proto);
}

static int spi_nor_spimem_check_readop(struct spi_nor *nor,
				       const struct spi_nor_read_command *read)
{
	struct spi_mem_op op = SPI_NOR_READ_OP(read->opcode);

	spi_nor_spimem_setup_op(nor, &op, read->proto);

	/* convert the dummy cycles to the number of bytes */
	op.dummy.nbytes = (read->num_mode_clocks + read->num_wait_states) *
			  op.dummy.buswidth / 8;

	return spi_nor_spimem_check_op(nor, &op);
}
```

​	第29行先简单初始化一个`op`参数，第31行根据读传输协议初始化`op`参数中各个`buswidth`参数，比如，如果是`1-1-4`读传输协议，则`op->cmd.buswidth = 1`、`op->addr.buswidth = 1`、`op->dummy.buswidth = 1`、`op->data.buswidth = 4`，第33-35行初始化`dummy`参数。

3. `spi_nor_spimem_check_op()`函数如下：

```c
static int spi_nor_spimem_check_op(struct spi_nor *nor,
				   struct spi_mem_op *op)
{
	/*
	 * First test with 4 address bytes. The opcode itself might
	 * be a 3B addressing opcode but we don't care, because
	 * SPI controller implementation should not check the opcode,
	 * but just the sequence.
	 */
	op->addr.nbytes = 4;
	if (!spi_mem_supports_op(nor->spimem, op)) {
		if (nor->params->size > SZ_16M)
			return -EOPNOTSUPP;

		/* If flash size <= 16MB, 3 address bytes are sufficient */
		op->addr.nbytes = 3;
		if (!spi_mem_supports_op(nor->spimem, op))
			return -EOPNOTSUPP;
	}

	return 0;
}
```

​	从代码来看，会进行两次校验，第一次是4字节地址校验，第二次是3字节地址校验。但是实际上并不会对`op->cmd.opcode`和`op->addr.nbytes`做是否匹配的校验，即没有"当读指令为3字节地址指令，而`op->addr.nbytes`值为4时，返回`false`"的条件判断。如果第11行判断为真（即`op`不合法），则第17行判断也为真，那就会做两次`op`校验；如果第17行判断为假（即`op`合法），则第11行判断也为假，那就不会跑第17行代码。

​	在注释里解释的很清楚，第一次测试有4个地址字节。操作码本身可能是3字节寻址操作码，但我们不在乎，因为`SPI`控制器实现不应该检查操作码，而只是检查序列。

> ​	此处笔者也没有看懂为什么需要做两次校验，起初怀疑是有些`spi`控制器会检查操作码和`op->addr.nbytes`是否相匹配，如果不匹配则说明`op`不合法，匹配则说明合法。这样的话做两次校验就能解释的通了。但是注释中又说“`SPI`控制器实现不应该检查操作码”。此处笔者还是留有疑问，后续有合理解释再进行补充。

4. `spi_mem_supports_op()`函数及其调用函数如下：

```c
static int spi_mem_check_op(const struct spi_mem_op *op)
{
	if (!op->cmd.buswidth || !op->cmd.nbytes)
		return -EINVAL;

	if ((op->addr.nbytes && !op->addr.buswidth) ||
	    (op->dummy.nbytes && !op->dummy.buswidth) ||
	    (op->data.nbytes && !op->data.buswidth))
		return -EINVAL;

	if (!spi_mem_buswidth_is_valid(op->cmd.buswidth) ||
	    !spi_mem_buswidth_is_valid(op->addr.buswidth) ||
	    !spi_mem_buswidth_is_valid(op->dummy.buswidth) ||
	    !spi_mem_buswidth_is_valid(op->data.buswidth))
		return -EINVAL;

	/* Buffers must be DMA-able. */
	if (WARN_ON_ONCE(op->data.dir == SPI_MEM_DATA_IN &&
			 object_is_on_stack(op->data.buf.in)))
		return -EINVAL;

	if (WARN_ON_ONCE(op->data.dir == SPI_MEM_DATA_OUT &&
			 object_is_on_stack(op->data.buf.out)))
		return -EINVAL;

	return 0;
}

static bool spi_mem_internal_supports_op(struct spi_mem *mem,
					 const struct spi_mem_op *op)
{
	struct spi_controller *ctlr = mem->spi->controller;

	if (ctlr->mem_ops && ctlr->mem_ops->supports_op)
		return ctlr->mem_ops->supports_op(mem, op);

	return spi_mem_default_supports_op(mem, op);
}

bool spi_mem_supports_op(struct spi_mem *mem, const struct spi_mem_op *op)
{
	if (spi_mem_check_op(op))
		return false;

	return spi_mem_internal_supports_op(mem, op);
}
```

​	第42行，`spi_mem_check_op()`函数主要是校验`op`中的`buswidth`以及缓冲区是否支持`DMA`。第45行，`spi_mem_internal_supports_op()`函数对`op`做进一步校验。

​	一般来说，`qspi`控制器一般都会实现`ctlr->mem_ops->supports_op`方法，所以一般情况下并不会跑到第37行。

```c
				     const struct spi_mem_op *op)

	ret = phytium_qspi_check_buswidth(op->cmd.buswidth);

		ret |= phytium_qspi_check_buswidth(op->addr.buswidth);

		return false;
```

6. `spi_mem_default_supports_op()`函数及其调用函数定义如下：

```c
static bool spi_mem_check_buswidth(struct spi_mem *mem,
				   const struct spi_mem_op *op)
{
	if (spi_check_buswidth_req(mem, op->cmd.buswidth, true))
		return false;

	if (op->addr.nbytes &&
	    spi_check_buswidth_req(mem, op->addr.buswidth, true))
		return false;

	if (op->dummy.nbytes &&
	    spi_check_buswidth_req(mem, op->dummy.buswidth, true))
		return false;

	if (op->data.dir != SPI_MEM_NO_DATA &&
	    spi_check_buswidth_req(mem, op->data.buswidth,
				   op->data.dir == SPI_MEM_DATA_OUT))
		return false;

	return true;
}

bool spi_mem_default_supports_op(struct spi_mem *mem,
				 const struct spi_mem_op *op)
{
	if (op->cmd.dtr || op->addr.dtr || op->dummy.dtr || op->data.dtr)
		return false;

	if (op->cmd.nbytes != 1)
		return false;

	return spi_mem_check_buswidth(mem, op);
}
```

​	第26行，前面没有对`op`中的`dtr`参数进行过赋值，所以其为`NULL`，该行判断为假。第29行，对`op`的`cmd`进行校验，`cmd`只能是1个字节。第32行通过`spi_mem_check_buswidth()`函数对`op`中的各个`buswidth`再做进一步校验。

> ​	笔者认为第4、8、12行的`true`都应该改成`op->data.dir == SPI_MEM_DATA_OUT`，这样并不会影响正常测试，而且逻辑上也更自洽。否则可能会出现以下情况：设备树中的`spi-rx-bus-width = <2>`和`spi-tx-bus-width = <1>`，在校验`1-2-2`读传输协议时，第8行判断为真，则`1-2-2`读传输协议不合法。但是设备树中的`spi-rx-bus-width`的值为2，这样看`1-2-2`读传输协议应该是合法的，显然矛盾。
>
> ​	需要注意，上面所说 的`1-2-2`读传输协议仅用做举例，实际上在第`2.1`节只会设置`1-1-x`读传输协议，并不会设置`1-2-2`读传输协议。

```c
static int spi_check_buswidth_req(struct spi_mem *mem, u8 buswidth, bool tx)
{
	u32 mode = mem->spi->mode;

	switch (buswidth) {
	case 1:
		return 0;

	case 2:
		if ((tx &&
		     (mode & (SPI_TX_DUAL | SPI_TX_QUAD | SPI_TX_OCTAL))) ||
		    (!tx &&
		     (mode & (SPI_RX_DUAL | SPI_RX_QUAD | SPI_RX_OCTAL))))
			return 0;

		break;

	case 4:
		if ((tx && (mode & (SPI_TX_QUAD | SPI_TX_OCTAL))) ||
		    (!tx && (mode & (SPI_RX_QUAD | SPI_RX_OCTAL))))
			return 0;

		break;

	case 8:
		if ((tx && (mode & SPI_TX_OCTAL)) ||
		    (!tx && (mode & SPI_RX_OCTAL)))
			return 0;

		break;

    default:
		break;
    }

	return -ENOTSUPP;
}
```
